## 关键实现要点

### ⭐ `src/services/scrub.js` —— 双层防御的第二层
```js
const BLACKLIST = new Set([
  "name","realName","studentNo","studentId","stuNo","teacher","teacherName",
  "className","major","college","grade","score","gpa","rank","phone","mobile","idCard",
]);
const ALLOWLIST = new Set(["weekday","period_start","period_end","title","room","weeks"]);

/**
 * 返回 { clean: [...], dropped: ["name","teacher"] }
 * ⛔ dropped 只记字段名，绝不记值
 */
export function scrubScheduleItems(items) {
  const dropped = new Set();
  const clean = [];
  for (const raw of items) {
    const item = {};
    for (const [k, v] of Object.entries(raw)) {
      const key = String(k);
      if (BLACKLIST.has(key)) { dropped.add(key); continue; }   // 命中黑名单 → 丢
      if (!ALLOWLIST.has(key)) { dropped.add(key); continue; }  // 不在白名单 → 丢
      item[key] = v;
    }
    // 必填校验
    if (item.weekday == null || item.period_start == null
        || item.period_end == null || !item.title) {
      dropped.add(item.title ? "incomplete_row" : "title");
      continue;
    }
    if (item.period_end < item.period_start) { dropped.add("bad_period"); continue; }
    clean.push(item);
  }
  return { clean, dropped: [...dropped] };
}

export function writeAudit(db, itemType, droppedFields) {
  if (!droppedFields.length) return;
  db.prepare(
    "INSERT INTO scrub_audit(item_type, dropped_fields, created_at) VALUES (?,?,?)"
  ).run(itemType, JSON.stringify(droppedFields), Math.floor(Date.now() / 1000));
}
```

**注意**：`title` 若含教师姓名（如"张三老师的选修"），在写入前去掉人名部分。
名字里的禁字段**值**同样不许进 `scrub_audit`。
