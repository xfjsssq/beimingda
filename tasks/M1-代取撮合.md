## 关键实现要点

### 撮合引擎必须返回可解释理由
```js
// engine/match.js
export function matchScore(task, runner, now) {
  // 1) 可达性
  const reach = JSON.parse(runner.reachable_buildings);
  let accessibility = 0;
  if (reach.includes(task.to_building))        accessibility = 40;
  else if (reach.includes(task.from_building)) accessibility = 20;
  else return null;                       // 出局

  // 2) 时间
  const wins = JSON.parse(runner.time_windows);
  let timeScore = 0;
  for (const w of wins) {
    const overlap = Math.max(0, Math.min(w.end, task.deadline_ts) - Math.max(w.start, now));
    const span    = task.deadline_ts - now;
    if (span <= 0) continue;
    if (overlap >= span && w.start <= now) { timeScore = 30; break; }
    const ratio = overlap / span;
    if (ratio >= 0.5) timeScore = Math.max(timeScore, 20);
    else if (ratio > 0) timeScore = Math.max(timeScore, 10);
  }
  if (timeScore === 0) return null;       // 完全不重叠 → 出局

  // 3) 报酬
  const ratio = Math.min(1, Math.max(0,
    (task.reward_max - runner.reward_floor) /
    (task.reward_max - task.reward_min + 1e-6)));
  const rewardScore = ratio * 30;

  const score = accessibility + timeScore + rewardScore;
  const reason = `可达 ${task.to_building}(+${accessibility})，`
               + `时段覆盖(+${timeScore})，`
               + `报酬 ${task.reward_max}元 vs 下限 ${runner.reward_floor}元(+${rewardScore.toFixed(0)})`;
  return { score, reason };
}
```
